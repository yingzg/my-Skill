```python
"""解析 .flow.yaml 文件，输出 FlowConfig 对象。"""

from __future__ import annotations

import yaml
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


class ParseError(Exception):
    """YAML 解析或校验错误。"""

    pass


@dataclass
class AuthConfig:
    type: str  # oauth2 | bearer | apikey | none
    token_url: str = ""
    extra: dict[str, str] = field(default_factory=dict)


@dataclass
class Dependency:
    table: str
    reason: str = ""
    fields: list[str] = field(default_factory=list)


@dataclass
class SuspectedBranch:
    source: str
    type: str = ""
    hint: str = ""


@dataclass
class Request:
    method: str
    url: str
    headers: dict[str, str] = field(default_factory=dict)
    body: Any = None


@dataclass
class TestCase:
    id: str
    scenario: str
    priority: str  # high | medium | low
    prepare: list[str]
    request: Request
    branch: str = ""
    expect_error: bool = False
    extract: dict[str, str] = field(default_factory=dict)
    cleanup: list[str] = field(default_factory=list)


@dataclass
class FlowConfig:
    name: str
    description: str
    base_url: str
    auth: AuthConfig
    test_cases: list[TestCase]
    dependencies: list[Dependency] = field(default_factory=list)
    ignore_response_fields: list[str] = field(default_factory=list)
    suspected_branches: list[SuspectedBranch] = field(default_factory=list)
    global_cleanup: list[str] = field(default_factory=list)


def load_flow(path: Path | str) -> FlowConfig:
    """从 YAML 文件加载 FlowConfig，含校验。"""
    path = Path(path)
    if not path.exists():
        raise ParseError(f"文件不存在: {path}")

    with open(path, "r", encoding="utf-8") as f:
        raw = yaml.safe_load(f)

    if not raw or not isinstance(raw, dict):
        raise ParseError("YAML 内容为空或格式错误")

    # 必填字段校验
    for field_name in ("name", "base_url", "auth", "test_cases"):
        if field_name not in raw:
            raise ParseError(f"缺少必填字段: {field_name}")

    # 解析 auth
    auth_raw = raw["auth"] or {}
    auth = AuthConfig(
        type=auth_raw.get("type", "none"),
        token_url=auth_raw.get("token_url", ""),
        extra={k: v for k, v in auth_raw.items() if k not in ("type", "token_url")},
    )

    # 解析 test_cases
    test_cases: list[TestCase] = []
    tc_ids: set[str] = set()

    for i, tc_raw in enumerate(raw["test_cases"]):
        tc_id = tc_raw.get("id", f"tc-{i:03d}")
        if tc_id in tc_ids:
            raise ParseError(f"重复的 test case id: {tc_id}")
        tc_ids.add(tc_id)

        req_raw = tc_raw.get("request", {})
        request = Request(
            method=req_raw.get("method", "GET").upper(),
            url=req_raw.get("url", ""),
            headers=req_raw.get("headers") or {},
            body=req_raw.get("body"),
        )

        tc = TestCase(
            id=tc_id,
            scenario=tc_raw.get("scenario", ""),
            branch=tc_raw.get("branch", ""),
            priority=tc_raw.get("priority", "medium"),
            expect_error=tc_raw.get("expect_error", False),
            prepare=tc_raw.get("prepare") or [],
            request=request,
            extract=tc_raw.get("extract") or {},
            cleanup=tc_raw.get("cleanup") or [],
        )
        test_cases.append(tc)

    # 解析 dependencies
    dependencies = [
        Dependency(
            table=d.get("table", ""),
            reason=d.get("reason", ""),
            fields=d.get("fields", []),
        )
        for d in raw.get("dependencies", [])
    ]

    # 解析 suspected_branches
    suspected_branches = [
        SuspectedBranch(
            source=sb.get("source", ""),
            type=sb.get("type", ""),
            hint=sb.get("hint", ""),
        )
        for sb in raw.get("suspected_branches", [])
    ]

    return FlowConfig(
        name=raw["name"],
        description=raw.get("description", ""),
        base_url=raw["base_url"],
        auth=auth,
        dependencies=dependencies,
        ignore_response_fields=raw.get("ignore_response_fields", []),
        suspected_branches=suspected_branches,
        test_cases=test_cases,
        global_cleanup=raw.get("global_cleanup", []),
    )
```

"""运行时变量管理 + {{@var}} 模板替换。"""

from __future__ import annotations

import re
from typing import Any

_VAR_PATTERN = re.compile(r"\{\{([@A-Z][A-Za-z0-9_]*)\}\}")


class RuntimeVariables:
    """存储 prepare SQL 产生的变量，供 request body 和 cleanup SQL 引用。"""

    def __init__(self) -> None:
        self._vars: dict[str, str] = {}

    def set(self, name: str, value: str) -> None:
        self._vars[name] = value

    def get(self, name: str) -> str | None:
        return self._vars.get(name)

    def set_auth_token(self, token: str) -> None:
        self._vars["AUTH_TOKEN"] = token

    def items(self):
        return self._vars.items()


def substitute(obj: Any, runtime: RuntimeVariables) -> Any:
    """递归替换对象中所有 {{var}} 占位符为 runtime 中的值。

    Args:
        obj: 字符串、字典、列表或基础类型
        runtime: 包含变量值的 RuntimeVariables 实例

    Returns:
        替换后的对象（类型与输入相同）

    Raises:
        KeyError: 存在 {{@var}} 但 runtime 中没有对应的变量
    """
    if isinstance(obj, str):
        result = obj
        for match in _VAR_PATTERN.finditer(obj):
            var_name = match.group(1)
            value = runtime.get(var_name)
            if value is None:
                raise KeyError(
                    f"变量 '{var_name}' 未在 runtime_vars 中找到。"
                    f" 可用变量: {list(runtime.items())}"
                )
            result = result.replace(match.group(0), value)
        return result
    elif isinstance(obj, dict):
        return {k: substitute(v, runtime) for k, v in obj.items()}
    elif isinstance(obj, list):
        return [substitute(item, runtime) for item in obj]
    else:
        return obj


def parse_set_statement(sql: str) -> tuple[str, str] | None:
    """解析 SET @var = ... 语句，返回 (变量名, 值)。

    只处理简单形式: SET @varname = 'literal_value';
    不处理表达式——LAST_INSERT_ID() 等由 MySQL 端解析。
    """
    m = re.match(
        r"^\s*SET\s+(@[A-Za-z0-9_]+)\s*=\s*(.+?);?\s*$", sql, re.IGNORECASE
    )
    if m:
        var_name = m.group(1)
        expr = m.group(2).rstrip(";").strip()
        return var_name, expr
    return None
```

"""MySQL 数据库操作：prepare 和 cleanup SQL 执行。"""

from __future__ import annotations

import re
from typing import Any

import pymysql
from api_flow.variables import RuntimeVariables

_SET_RE = re.compile(
    r"^\s*SET\s+(@[A-Za-z0-9_]+)\s*=\s*(.+?);?\s*$", re.IGNORECASE
)


def parse_set_var_from_sql(sql: str) -> tuple[str, str] | None:
    """从 SET @var = expr 语句中提取变量名和表达式。"""
    m = _SET_RE.match(sql.strip())
    if m:
        return m.group(1), m.group(2).rstrip(";").strip()
    return None


class Database:
    """MySQL 数据库连接封装。"""

    def __init__(self, host: str, user: str, password: str, database: str, port: int = 3306) -> None:
        self._config = {
            "host": host,
            "user": user,
            "password": password,
            "database": database,
            "port": port,
            "charset": "utf8mb4",
        }
        self._conn: pymysql.Connection | None = None

    def _connect(self) -> pymysql.Connection:
        if self._conn is None or not self._conn.open:
            self._conn = pymysql.connect(**self._config)
        return self._conn

    def execute_prepare(self, sql_statements: list[str], runtime: RuntimeVariables) -> None:
        """执行 prepare SQL，捕获 SET @var 语句中的变量到 runtime。

        对每条 SET @var = expr 语句，执行 SELECT expr AS val 来获取实际值。
        """
        conn = self._connect()
        with conn.cursor() as cursor:
            for stmt in sql_statements:
                stmt = stmt.strip()
                if not stmt:
                    continue

                parsed = parse_set_var_from_sql(stmt)
                if parsed:
                    var_name, expr = parsed
                    # 执行 SELECT <expr> 获取实际值
                    cursor.execute(f"SELECT {expr} AS val")
                    row = cursor.fetchone()
                    if row:
                        runtime.set(var_name, str(row[0]))
                else:
                    cursor.execute(stmt)

        conn.commit()

    def execute_cleanup(self, sql_statements: list[str]) -> None:
        """执行 cleanup SQL（变量已由调用方替换）。"""
        conn = self._connect()
        with conn.cursor() as cursor:
            for stmt in sql_statements:
                stmt = stmt.strip()
                if stmt:
                    cursor.execute(stmt)
        conn.commit()

    def close(self) -> None:
        if self._conn and self._conn.open:
            self._conn.close()

    def __enter__(self):
        return self

    def __exit__(self, *args: Any) -> None:
        self.close()
```

from dataclasses import dataclass
from pathlib import Path

import requests


class AuthError(Exception):
    """鉴权失败。"""
    pass


@dataclass
class AuthConfig:
    type: str  # oauth2 | bearer | apikey | none
    token_url: str = ""
    token: str = ""        # bearer 固定 token

class AuthInjector:
    """处理鉴权获取和 token 注入。"""

    def __init__(self, config: AuthConfig) -> None:
        self._config = config
        self._cached_token: str | None = None

    def get_token(self) -> str:
        """获取 token（带缓存）。"""
        if self._cached_token is not None:
            return self._cached_token

        token = self._fetch_token()
        self._cached_token = token
        return token

    def _fetch_token(self) -> str:
        t = self._config.type
        if t == "none":
            return ""
        elif t == "bearer":
            token = self._config.token
            if not token:
                raise AuthError("bearer 类型需要提供 token")
            return f"Bearer {token}"
        elif t == "apikey":
            return self._config.api_key
        elif t == "oauth2":
            return self._fetch_oauth2()
        else:
            raise AuthError(f"不支持的鉴权类型: {t}")

    def _fetch_oauth2(self) -> str:
        cfg = self._config
        try:
            resp = requests.post(
                cfg.token_url,
                data={
                    "grant_type": "password",
                    "client_id": cfg.client_id,
                    "client_secret": cfg.client_secret,
                    "username": cfg.username,
                    "password": cfg.password,
                },
                timeout=30,
            )
            resp.raise_for_status()
            data = resp.json()
            access_token = data.get("access_token", "")
            if not access_token:
                raise AuthError("响应中未找到 access_token")
            return f"Bearer {access_token}"
        except requests.RequestException as e:
            raise AuthError(f"获取 OAuth2 token 失败: {e}") from e

        api_key=os.environ.get("API_FLOW_API_KEY", raw.get("api_key", "")),

"""HTTP 请求发送 + JSONPath 响应提取。"""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from typing import Any

import requests
from jsonpath_rw import parse as jsonpath_parse

from api_flow.variables import RuntimeVariables, substitute


@dataclass
class HttpResponse:
    status_code: int
    headers: dict[str, str]
    body: Any
    text: str


class HttpClient:
    """发送 HTTP 请求，自动替换 {{@var}} 变量，支持 JSONPath 提取。"""

    def __init__(self, base_url: str, timeout: int = 30) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout
        self._session = requests.Session()

    def request(
        self,
        method: str,
        url: str,
        headers: dict[str, str] | None = None,
        body: Any = None,
        runtime: RuntimeVariables | None = None,
    ) -> HttpResponse:
        """发送一个 HTTP 请求。变量替换在发送前完成。"""
        full_url = f"{self._base_url}{url}"

        if runtime:
            headers = substitute(headers or {}, runtime)
            body = substitute(body, runtime)

        resp = self._session.request(
            method=method.upper(),
            url=full_url,
            headers=headers or {},
            json=body,
            timeout=self._timeout,
        )

        try:
            resp_body = resp.json()
        except (json.JSONDecodeError, ValueError):
            resp_body = resp.text

        return HttpResponse(
            status_code=resp.status_code,
            headers=dict(resp.headers),
            body=resp_body,
            text=resp.text,
        )

    def extract_to_runtime(
        self,
        response_body: Any,
        extract_rules: dict[str, str],
        runtime: RuntimeVariables,
    ) -> None:
        """从响应 body 中按 JSONPath 提取值，写入 runtime。

        extract_rules 格式: {"varname": "$.path.to.field"}
        提取后存入 runtime: @varname = "extracted_value"
        """
        for var_name, jsonpath_expr in extract_rules.items():
            try:
                expr = jsonpath_parse(jsonpath_expr)
                matches = expr.find(response_body)
                if matches:
                    value = str(matches[0].value)
                    runtime.set(f"@{var_name}", value)
            except Exception:
                # JSONPath 解析失败，跳过（不阻塞执行）
                pass

    def close(self) -> None:
        self._session.close()
```

"""录制输出：将每次执行的结果保存为结构化 JSON 文件。"""

from __future__ import annotations

import json
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from typing import Any


@dataclass
class TResult:
    tc_id: str
    scenario: str
    status: str  # pass | fail | skipped | error
    status_code: int | None = None
    duration_ms: float = 0.0
    error: str = ""
    request_body: Any = None
    response_body: Any = None
    prepare_sql: list[str] | None = None
    cleanup_sql: list[str] | None = None


class Recorder:
    """管理 api-flow-output 目录结构并写入结果。"""

    def __init__(self, flow_name: str, output_root: str = "api-flow-output") -> None:
        timestamp = datetime.now().strftime("%Y%m%d-%H%M%S")
        self._run_dir = Path(output_root) / flow_name / f"run-{timestamp}"
        self._run_dir.mkdir(parents=True, exist_ok=True)

    def record(
        self,
        result: TResult,
    ) -> None:
        """保存单个测试用例的执行结果。"""
        tc_dir = self._run_dir / result.tc_id
        tc_dir.mkdir(exist_ok=True)

        if result.prepare_sql:
            (tc_dir / "prepare.sql").write_text("\n".join(result.prepare_sql), encoding="utf-8")

        if result.request_body is not None:
            (tc_dir / "request.json").write_text(
                json.dumps(result.request_body, ensure_ascii=False, indent=2),
                encoding="utf-8",
            )

        if result.response_body is not None:
            (tc_dir / "response.json").write_text(
                json.dumps(result.response_body, ensure_ascii=False, indent=2),
                encoding="utf-8",
            )

        if result.cleanup_sql:
            (tc_dir / "cleanup.sql").write_text("\n".join(result.cleanup_sql), encoding="utf-8")

        result_obj = {
            "tc_id": result.tc_id,
            "scenario": result.scenario,
            "status": result.status,
            "status_code": result.status_code,
            "duration_ms": result.duration_ms,
            "error": result.error,
        }
        (tc_dir / "result.json").write_text(
            json.dumps(result_obj, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    def save_summary(self, results: list[TResult]) -> Path:
        """保存汇总 JSON。"""
        passed = sum(1 for r in results if r.status == "pass")
        failed = sum(1 for r in results if r.status == "fail")
        skipped = sum(1 for r in results if r.status == "skipped")

        summary = {
            "flow_name": self._run_dir.parent.name,
            "run_id": self._run_dir.name,
            "total": len(results),
            "passed": passed,
            "failed": failed,
            "skipped": skipped,
            "results": [
                {
                    "tc_id": r.tc_id,
                    "scenario": r.scenario,
                    "status": r.status,
                    "status_code": r.status_code,
                    "duration_ms": r.duration_ms,
                    "error": r.error,
                }
                for r in results
            ],
        }
        summary_path = self._run_dir / "summary.json"
        summary_path.write_text(
            json.dumps(summary, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )
        return summary_path

    @property
    def run_dir(self) -> Path:
        return self._run_dir
```

"""执行协调器：prepare → HTTP request → cleanup 主循环。"""

from __future__ import annotations

import time
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from api_flow.parser import FlowConfig, load_flow, TestCase
from api_flow.variables import RuntimeVariables, substitute
from api_flow.auth import AuthInjector, auth_config_from_yaml
from api_flow.db import Database
from api_flow.http_client import HttpClient, HttpResponse
from api_flow.recorder import Recorder, TResult


@dataclass
class RunResult:
    total: int = 0
    passed: int = 0
    failed: int = 0
    skipped: int = 0
    results: list[TResult] = field(default_factory=list)


class Engine:
    """api-flow 执行引擎。"""

    def __init__(
        self,
        flow_path: Path | str,
        output_root: str = "api-flow-output",
        fail_fast: bool = False,
        db_config: dict[str, Any] | None = None,
    ) -> None:
        self._flow: FlowConfig = load_flow(flow_path)
        self._output_root = output_root
        self._fail_fast = fail_fast
        self._db_config = db_config or {}

    def run(self) -> RunResult:
        """执行全部测试用例。"""
        result = RunResult()
        result.total = len(self._flow.test_cases)

        # 初始化组件
        auth_injector = AuthInjector(auth_config_from_yaml({
            "type": self._flow.auth.type,
            "token_url": self._flow.auth.token_url,
            **self._flow.auth.extra,
        }))
        db = Database(
            host=self._db_config.get("host", "localhost"),
            user=self._db_config.get("user", "root"),
            password=self._db_config.get("password", ""),
            database=self._db_config.get("database", "test"),
            port=self._db_config.get("port", 3306),
        )
        http_client = HttpClient(base_url=self._flow.base_url)
        recorder = Recorder(self._flow.name, self._output_root)

        try:
            for tc in self._flow.test_cases:
                try:
                    tc_result = self._run_one(tc, auth_injector, db, http_client, recorder)
                    result.results.append(tc_result)

                    if tc_result.status == "pass":
                        result.passed += 1
                    elif tc_result.status == "fail":
                        result.failed += 1
                        if self._fail_fast:
                            break
                    else:
                        result.skipped += 1

                except Exception as e:
                    tr = TResult(
                        tc_id=tc.id,
                        scenario=tc.scenario,
                        status="error",
                        error=str(e),
                    )
                    result.results.append(tr)
                    result.failed += 1
                    if self._fail_fast:
                        break
        finally:
            recorder.save_summary(result.results)
            http_client.close()
            db.close()

        return result

    def _run_one(
        self,
        tc: TestCase,
        auth_injector: AuthInjector,
        db: Database,
        http_client: HttpClient,
        recorder: Recorder,
    ) -> TResult:
        """执行单个测试用例。"""
        tr = TResult(tc_id=tc.id, scenario=tc.scenario, status="skipped")
        runtime = RuntimeVariables()

        # 注入 auth token
        token = auth_injector.get_token()
        runtime.set_auth_token(token)

        start = time.time()

        # Phase 1: Prepare
        try:
            db.execute_prepare(tc.prepare, runtime)
            tr.prepare_sql = tc.prepare
        except Exception as e:
            tr.status = "error"
            tr.error = f"prepare 失败: {e}"
            tr.duration_ms = (time.time() - start) * 1000
            recorder.record(tr)
            return tr

        # Phase 2: Run
        try:
            response = http_client.request(
                method=tc.request.method,
                url=tc.request.url,
                headers=tc.request.headers,
                body=tc.request.body,
                runtime=runtime,
            )
            tr.status_code = response.status_code
            tr.request_body = tc.request.body
            tr.response_body = response.body
            tr.duration_ms = (time.time() - start) * 1000

            # 判断 pass/fail
            if tc.expect_error:
                tr.status = "pass" if response.status_code >= 400 else "fail"
            else:
                tr.status = "pass" if 200 <= response.status_code < 300 else "fail"

            # Extract 响应变量
            http_client.extract_to_runtime(response.body, tc.extract, runtime)

        except Exception as e:
            tr.status = "error"
            tr.error = f"请求失败: {e}"
            tr.duration_ms = (time.time() - start) * 1000
            recorder.record(tr)
            return tr

        # Phase 3: Cleanup
        try:
            # 替换 cleanup SQL 中的变量
            substituted_cleanup = [substitute(stmt, runtime) for stmt in tc.cleanup]
            tr.cleanup_sql = substituted_cleanup
            db.execute_cleanup(substituted_cleanup)
        except Exception as e:
            # cleanup 失败不改变 test case 结果，但记录 error
            if tr.status == "pass":
                pass  # 主测试已通过，cleanup 失败单独记录

        recorder.record(tr)
        return tr
```

"""报告生成：终端表格 + 汇总输出。"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

STATUS_ICONS = {
    "pass": "\u2705",   # ✅
    "fail": "\u274c",   # ❌
    "error": "\u274c",  # ❌
    "skipped": "\u26a0\ufe0f",  # ⚠️
}


@dataclass
class RunResult:
    total: int = 0
    passed: int = 0
    failed: int = 0
    skipped: int = 0
    results: list[dict[str, Any]] = field(default_factory=list)


def print_execution_plan(flow_name: str, total: int, high: int, medium: int) -> None:
    """打印执行计划概览。"""
    print(f"\n\U0001f4cb {flow_name} - 测试计划")
    print(f"用例数: {total}  高优: {high}  中优: {medium}")
    print("执行模式: 每组独立（prepare → run → 回滚）\n")


def print_case_result(
    tc_id: str, scenario: str, status: str, status_code: int | None,
    duration_ms: float, error: str = "",
) -> None:
    """打印单个测试用例结果。"""
    icon = STATUS_ICONS.get(status, "❓")
    sc = f"[{status_code}]" if status_code else ""
    print(f"  {icon} {tc_id} {sc}  {scenario}")

    if status == "fail" or status == "error":
        if error:
            print(f"     错误: {error}")
        if status_code:
            print(f"     HTTP status: {status_code}")
    if duration_ms > 0:
        print(f"     耗时: {duration_ms:.0f}ms")


def print_summary(flow_name: str, result: RunResult) -> None:
    """打印测试汇总。"""
    print(f"\n\U0001f4ca {flow_name} - 测试汇总")
    print(f"总数: {result.total}  "
          f"通过: {result.passed}  "
          f"失败: {result.failed}  "
          f"跳过: {result.skipped}")

    failures = [r for r in result.results if r.get("status") not in ("pass",)]
    if failures:
        print("\n失败/错误用例:")
        for r in failures:
            icon = STATUS_ICONS.get(r.get("status", ""), "❓")
            sc = f"[{r.get('status_code')}]" if r.get("status_code") else ""
            err = f" - {r.get('error')}" if r.get("error") else ""
            print(f"  {icon} {r['tc_id']} {sc}{err}")
```

from pathlib import Path

from api_flow.reporter import print_execution_plan, print_case_result, print_summary, RunResult

def cmd_run(args: argparse.Namespace) -> int:
    """执行 .flow.yaml 中的全部测试用例。"""
    flow_path = Path(args.file)
    if not flow_path.exists():
        print(f"❌ 文件不存在: {flow_path}", file=sys.stderr)
        return 1

    engine = Engine(
        flow_path=flow_path,
        output_root=args.output,
        fail_fast=args.fail_fast,
        db_config={
            "host": args.db_host,
            "user": args.db_user,
            "password": args.db_password,
            "database": args.db_name,
            "port": args.db_port,
        },
    )
    flow = engine._flow

    # 打印执行计划
    high = sum(1 for tc in flow.test_cases if tc.priority == "high")
    medium = sum(1 for tc in flow.test_cases if tc.priority == "medium")
    print_execution_plan(flow.name, len(flow.test_cases), high, medium)

    # 执行
    result = engine.run()

    # 逐用例打印结果
    for tr in result.results:
        print_case_result(tr.tc_id, tr.scenario, tr.status, tr.status_code, tr.duration_ms, tr.error)

    # 汇总
    print_summary(flow.name, RunResult(
        total=result.total,
        passed=result.passed,
        failed=result.failed,
        skipped=result.skipped,
        results=[
            {"tc_id": r.tc_id, "status": r.status, "status_code": r.status_code, "error": r.error}
            for r in result.results
        ],
    ))

    return 0 if result.failed == 0 else 1

    return 0

if __name__ == "__main__":
    sys.exit(main())

# api-flow

自动化 API 流程测试工具：分析代码 → 生成测试用例 → 批量执行 → 数据清理。

## 核心概念

- **analyze**（AI 负责）：用 gitnexus 分析代码，识别分支，生成 .flow.yaml
- **run**（CLI 负责）：解析 .flow.yaml，逐用例执行 prepare → request → cleanup
- 每条用例独立执行，不共享数据，不互相污染

## 前置要求

1. 项目已安装 gitnexus
2. Python CLI 已安装：`cd skill/api-flow/cli && pip install -e ".[dev]"`
3. MySQL 数据库可访问（用于 prepare/cleanup）

---

## 工作流

### 阶段 1: 分析代码，生成 .flow.yaml

```
用户提供需要测试的 API 入口：
  例如: POST /api/orders  → Controller: com.example.controller.OrderController.createOrder()
  或直接: com.example.controller.OrderController.createOrder()
```

**AI 执行 analyze 流程：**

1. **发现入口** — gitnexus 搜索 Controller 类和方法
2. **追踪调用链** — 从 Controller → Service → Repository，最多 3 层
3. **识别分支** — 扫描 @Valid、if/else/switch、throw、自定义校验
4. **生成测试用例** — 每个分支一个用例，含中文场景描述
5. **生成 prepare SQL** — 分析分支条件反推所需数据，FK 按序排列
6. **生成 request body** — 复刻 DTO 结构，变量引用用 `{{@var}}`
7. **输出 .flow.yaml** — 保存到 `docs/api-flow/<name>.flow.yaml`

**.flow.yaml 格式：** 参考设计规格书 §2（`/root/docs/superpowers/specs/2026-07-16-api-flow-design.zh-CN.md`）

**局限性处理：**
- AOP / @EventListener / @Transactional → 不展开，写入 `suspected_branches`
- @Value / DictService 等配置读取 → 硬编码默认值 + SQL 注释提醒
- 调用链 > 3 层 → 截断，标记 `max_depth_exceeded`

### 阶段 2: 用户审阅 .flow.yaml

生成后提示用户：
1. 检查每个用例的 scenario 和 branch 是否正确
2. 检查 suspected_branches 是否有遗漏
3. 确认 prepare SQL 中的测试数据是否合适
4. 可手动调整变量名、priority、cleanup 等

### 阶段 3: 执行测试

```
api-flow run docs/api-flow/orders.flow.yaml \
  --db-host localhost \
  --db-user root \
  --db-password 123456 \
  --db-name mytestdb
```

CLI 自动：
1. 加载鉴权 → 注入 `{{AUTH_TOKEN}}`
2. 逐用例执行 prepare → request → cleanup
3. 录制输出到 `api-flow-output/<flow-name>/run-<timestamp>/`
4. 打印汇总报告

1. **先用一个简单接口试跑**，验证 .flow.yaml 格式和 CLI 工作正常
2. **高优先级用例先跑**：用 `priority: high` 标记核心业务路径
3. **expect_error 用例**：不要 prepare 数据（不需要），只验证返回了预期的错误码
4. **清理数据**：每个用例的 cleanup SQL 应该自包含，不依赖其他用例的数据
5. **鉴权信息**：敏感凭据通过项目 `docs/api-test/project-test-context.md` 管理
