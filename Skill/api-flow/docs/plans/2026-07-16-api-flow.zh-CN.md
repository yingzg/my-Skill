核心数据结构不变（`FlowConfig`, `TestCase`, `AuthConfig`, `RequestConfig`），变化在加载逻辑：

```python
"""flow 文件解析器：flow.yaml + JSON 用例 + SQL 脚本（v0.2 拆分结构）。"""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import yaml


# AuthConfig, RequestConfig, TestCase, FlowConfig 定义保持不变（略，见 v0.1 计划）
# 以下仅写新增/变更部分

class CaseRef:
    """.flow.yaml 中的单条用例引用。"""
    def __init__(self, id: str, file: str, prepare: str = "", cleanup: str = "") -> None:
        self.id = id
        self.file = file
        self.prepare = prepare
        self.cleanup = cleanup


def load_flow(flow_dir: Path | str) -> FlowConfig:
    """加载 flow 目录：解析 flow.yaml + 所有引用的用例文件。"""
    flow_dir = Path(flow_dir)
    flow_yaml_path = flow_dir / "flow.yaml"

    if not flow_yaml_path.exists():
        raise FileNotFoundError(f"flow.yaml 不存在: {flow_yaml_path}")

    raw = flow_yaml_path.read_text(encoding="utf-8")
    try:
        data = yaml.safe_load(raw)
    except yaml.YAMLError as e:
        raise ValueError(f"YAML 解析失败: {e}") from e

    # 解析鉴权
    auth_raw = data.get("auth", {})
    auth = AuthConfig(
        type=auth_raw.get("type", "none"),
        token_url=auth_raw.get("token_url", ""),
        extra={k: v for k, v in auth_raw.items() if k not in ("type", "token_url")},
    )

    # 解析用例引用 + 加载独立文件
    test_cases = []
    for ref_raw in data.get("cases", []):
        ref = CaseRef(
            id=ref_raw.get("id", ""),
            file=ref_raw.get("file", ""),
            prepare=ref_raw.get("prepare", ""),
            cleanup=ref_raw.get("cleanup", ""),
        )
        tc = _load_case_from_ref(flow_dir, ref)
        test_cases.append(tc)

    return FlowConfig(
        name=data.get("name", ""),
        base_url=data.get("base_url", ""),
        service=data.get("service", ""),
        auth=auth,
        test_cases=test_cases,
        suspected_branches=data.get("suspected_branches", []),
    )


def _load_case_from_ref(flow_dir: Path, ref: CaseRef) -> TestCase:
    """根据 flow.yaml 中的引用，加载独立 JSON 文件和 SQL 文件。"""
    # 加载 JSON
    json_path = flow_dir / ref.file
    if not json_path.exists():
        raise FileNotFoundError(f"用例文件不存在: {json_path}")

    case_data = json.loads(json_path.read_text(encoding="utf-8"))
    req_raw = case_data.get("request", {})

    req = RequestConfig(
        method=req_raw.get("method", "GET"),
        url=req_raw.get("url", ""),
        headers=req_raw.get("headers", {}),
        body=json.loads(json.dumps(req_raw.get("body", {}), default=str)),
    )

    # 加载 prepare SQL 文件（可选）
    prepare_sql: list[str] = []
    if ref.prepare:
        prep_path = flow_dir / ref.prepare
        if prep_path.exists():
            raw_sql = prep_path.read_text(encoding="utf-8")
            prepare_sql = _parse_sql_statements(raw_sql)

    # 加载 cleanup SQL 文件（可选）
    cleanup_sql: list[str] = []
    if ref.cleanup:
        clean_path = flow_dir / ref.cleanup
        if clean_path.exists():
            raw_sql = clean_path.read_text(encoding="utf-8")
            cleanup_sql = _parse_sql_statements(raw_sql)

    return TestCase(
        id=case_data.get("id", ref.id),
        scenario=case_data.get("scenario", ""),
        branch=case_data.get("branch", ""),
        priority=case_data.get("priority", "medium"),
        expect_error=case_data.get("expect_error", False),
        prepare=prepare_sql,
        extract=case_data.get("extract", {}),
        cleanup=cleanup_sql,
        request=req,
    )


def _parse_sql_statements(raw: str) -> list[str]:
    """将原始 SQL 文本拆分为独立语句（按 ; 分隔，过滤空行和纯注释）。"""
    statements = []
    for stmt in raw.split(";"):
        stmt = stmt.strip()
        if not stmt:
            continue
        # 跳过纯注释行
        if stmt.startswith("--"):
            continue
        statements.append(stmt + ";")
    return statements
```

```python
"""数据库连接：prepare / cleanup SQL 执行（v0.2: 新增 retry）。"""

from __future__ import annotations

import time
from typing import Any

import pymysql

from api_flow.variables import RuntimeVariables, substitute

# 可重试的 DB 错误码 / 异常类型
RETRYABLE_ERROR_CODES = {
    1205,   # ER_LOCK_WAIT_TIMEOUT — 锁等待超时
    1213,   # ER_LOCK_DEADLOCK — 死锁
    2002,   # CR_CONNECTION_ERROR — 连接错误
    2003,   # CR_CONN_HOST_ERROR — 无法连接主机
    2006,   # CR_SERVER_GONE_ERROR — 服务器断开
    2013,   # CR_SERVER_LOST — 查询期间断连
}
MAX_RETRIES = 3
RETRY_BACKOFF = [1, 2, 4]  # 秒


def _is_retryable(error: Exception) -> bool:
    """判断 DB 错误是否可重试。"""
    if isinstance(error, pymysql.err.OperationalError):
        if error.args and error.args[0] in RETRYABLE_ERROR_CODES:
            return True
    msg = str(error).lower()
    retryable_messages = [
        "connection refused", "connection reset", "connection timed out",
        "lost connection", "server has gone away", "deadlock", "lock wait timeout",
    ]
    return any(m in msg for m in retryable_messages)


class Database:
    """MySQL 数据库连接（通过 PyMySQL）。带指数退避重试。"""

    def __init__(
        self,
        host: str = "localhost",
        user: str = "root",
        password: str = "",
        database: str = "test",
        port: int = 3306,
    ) -> None:
        self._host = host
        self._user = user
        self._password = password
        self._database = database
        self._port = port
        self._conn: Any = None
        self._consecutive_errors: int = 0

    def connect(self) -> None:
        """建立数据库连接（autocommit 模式）。"""
        try:
            self._conn = pymysql.connect(
                host=self._host,
                user=self._user,
                password=self._password,
                database=self._database,
                port=self._port,
                autocommit=True,
                charset="utf8mb4",
            )
        except Exception as e:
            raise Exception(f"数据库连接失败: {e}") from e

    def execute_prepare(self, sql_statements: list[str], runtime: RuntimeVariables) -> None:
        """执行 prepare SQL——顺序执行，每句做变量替换。可重试错误自动退避重试。"""
        if not sql_statements:
            return
        self._execute_with_retry(sql_statements, runtime)

    def execute_cleanup(self, sql_statements: list[str], runtime: RuntimeVariables | None = None) -> None:
        """执行 cleanup SQL——顺序执行。不重试（清理失败直接忽略）。"""
        if not sql_statements:
            return
        if self._conn is None:
            self.connect()
        cursor = self._conn.cursor()
        try:
            for raw_sql in sql_statements:
                resolved = substitute(raw_sql, runtime) if runtime else raw_sql
                if not resolved.strip():
                    continue
                cursor.execute(resolved)
        finally:
            cursor.close()

    def _execute_with_retry(self, sql_statements: list[str], runtime: RuntimeVariables) -> None:
        """带指数退避重试的 SQL 执行。不可重试错误直接抛。"""
        last_error: Exception | None = None

        for attempt in range(MAX_RETRIES):
            try:
                self._execute_batch(sql_statements, runtime)
                self._consecutive_errors = 0  # 成功后重置
                return
            except Exception as e:
                last_error = e
                if not _is_retryable(e):
                    raise  # 不可重试：语法/权限错误，直接抛
                if attempt < MAX_RETRIES - 1:
                    time.sleep(RETRY_BACKOFF[attempt])
                    # 重试前重新连接
                    if self._conn:
                        try:
                            self._conn.close()
                        except Exception:
                            pass
                        self._conn = None
                    self.connect()

        self._consecutive_errors += 1
        raise Exception(f"SQL 执行失败（已重试 {MAX_RETRIES} 次）: {last_error}") from last_error

    def _execute_batch(self, sql_statements: list[str], runtime: RuntimeVariables) -> None:
        """批量执行 SQL（内部方法）。"""
        if self._conn is None:
            self.connect()
        cursor = self._conn.cursor()
        try:
            for raw_sql in sql_statements:
                resolved = substitute(raw_sql, runtime)
                if not resolved.strip():
                    continue
                cursor.execute(resolved)
        finally:
            cursor.close()

    @property
    def consecutive_errors(self) -> int:
        """连续失败计数（供 engine 判断是否全局中断）。"""
        return self._consecutive_errors

    def close(self) -> None:
        """关闭数据库连接。"""
        if self._conn:
            self._conn.close()
            self._conn = None
```

@patch("api_flow.db.pymysql.connect")
def test_retry_on_lock_timeout(self, mock_connect):
    """死锁第一次失败，第二次成功。"""
    mock_conn = MagicMock()
    mock_cursor = MagicMock()
    mock_cursor.execute.side_effect = [
        pymysql.err.OperationalError(1213, "Deadlock"),  # 第一次死锁
        None,  # 第二次成功
    ]
    mock_conn.cursor.return_value = mock_cursor
    mock_connect.return_value = mock_conn

    db = Database(**CONN_PARAMS)
    runtime = RuntimeVariables()
    db.execute_prepare(["INSERT INTO t VALUES (1);"], runtime)

    assert mock_cursor.execute.call_count == 2  # 重试了一次

@patch("api_flow.db.pymysql.connect")
def test_no_retry_on_syntax_error(self, mock_connect):
    """语法错误不应重试。"""
    mock_conn = MagicMock()
    mock_cursor = MagicMock()
    mock_cursor.execute.side_effect = pymysql.err.ProgrammingError(1064, "Syntax error")
    mock_conn.cursor.return_value = mock_cursor
    mock_connect.return_value = mock_conn

    db = Database(**CONN_PARAMS)
    runtime = RuntimeVariables()
    with pytest.raises(pymysql.err.ProgrammingError):
        db.execute_prepare(["BAD SQL SYNTAX!!!"], runtime)

STATUS_TEXT = {

    "fail": "失败",
    "error": "执行异常",
    "skipped": "跳过",
}


def render_case_result(
    tc_id: str,
    scenario: str,
    branch: str,
    priority: str,
    status: str,
    status_code: int | None,
    duration_ms: float,
    method: str,
    url: str,
    request_body: dict[str, Any] | None,
    response_body: Any,
    prepare_sql: list[str] | None,
    cleanup_sql: list[str] | None,
    error: str = "",
) -> str:
    """生成单条用例结果 Markdown（按设计规格书 §4.5 模板）。"""
    icon = STATUS_ICONS.get(status, "❓")
    status_text = STATUS_TEXT.get(status, status)

    req_body_str = json.dumps(request_body, ensure_ascii=False, indent=2) if request_body else ""
    resp_body_str = ""
    if response_body is not None:
        if isinstance(response_body, (dict, list)):
            resp_body_str = json.dumps(response_body, ensure_ascii=False, indent=2)
        else:
            resp_body_str = str(response_body)

    prepare_str = "\n".join(prepare_sql) if prepare_sql else "(无)"
    cleanup_str = "\n".join(cleanup_sql) if cleanup_sql else "(无)"

    return f"""## {tc_id} {icon} — {scenario}

| 项目 | 内容 |
|------|------|
| 分支 | {branch} |
| 优先级 | {priority} |
| 状态 | {icon} {status_text} |
| HTTP 状态码 | {status_code or "—"} |
| 耗时 | {duration_ms:.0f}ms |

### 请求

**{method or "GET"}** `{url or ""}`

```json
{req_body_str}
```

### 响应

```json
{resp_body_str}
```

### 数据操作

**Prepare SQL**（{len(prepare_sql) if prepare_sql else 0} 条）:
```sql
{prepare_str}
```

**Cleanup SQL**（{len(cleanup_sql) if cleanup_sql else 0} 条）:
```sql
{cleanup_str}
```
{f"### 错误信息\n\n```\n{error}\n```" if error else ""}
"""


def render_summary(
    flow_name: str,
    results: list[dict[str, Any]],
    total_duration_ms: float = 0,
) -> str:
    """生成汇总报告 Markdown（按设计规格书 §4.5 模板）。"""
    total = len(results)
    passed = sum(1 for r in results if r.get("status") == "pass")
    failed = sum(1 for r in results if r.get("status") in ("fail", "error"))
    skipped = total - passed - failed

    lines: list[str] = [
        f"# {flow_name} — 测试汇总",
        "",
        "| 项目 | 内容 |",
        "|------|------|",
        f"| 执行时间 | {datetime.now().strftime('%Y-%m-%d %H:%M:%S')} |",
        f"| 总用例数 | {total} |",
        f"| 通过 | {passed} |",
        f"| 失败 | {failed} |",
        f"| 跳过 | {skipped} |",
        f"| 总耗时 | {total_duration_ms:.0f}ms |",
        "",
        "## 用例明细",
        "",
        "| ID | 场景 | 状态 | HTTP | 耗时 |",
        "|----|------|------|------|------|",
    ]

    for r in results:
        icon = STATUS_ICONS.get(r.get("status", ""), "❓")
        sc = str(r.get("status_code", "—"))
        dur = f"{r.get('duration_ms', 0):.0f}ms"
        lines.append(
            f"| {r.get('tc_id', '')} "
            f"| {r.get('scenario', '')} "
            f"| {icon} "
            f"| {sc} "
            f"| {dur} |"
        )

    # 失败/异常汇总
    failures = [r for r in results if r.get("status") not in ("pass",)]
    if failures:
        lines.append("")
        lines.append("## 失败/异常")
        lines.append("")
        for r in failures:
            lines.append(f"### {r.get('tc_id', '')} — {r.get('scenario', '')}")
            lines.append(f"- 错误: {r.get('error', '未知错误')}")
            lines.append("")

    return "\n".join(lines)
```

```python
"""执行协调器：prepare → HTTP request → cleanup 主循环（v0.2: 新增 retry）。"""

from __future__ import annotations

import time
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from api_flow.parser import FlowConfig, load_flow, TestCase
from api_flow.variables import RuntimeVariables, substitute
from api_flow.auth import AuthInjector, auth_config_from_yaml
from api_flow.db import Database
from api_flow.http_client import HttpClient, HttpResponse
from api_flow.recorder import Recorder, TResult
from api_flow.templates import render_case_result, render_summary

MAX_CASE_RETRIES = 3  # 单条用例最多重试次数
MAX_CONSECUTIVE_FAILURES = 3  # 连续失败 N 条用例后全局中断


@dataclass
class RunResult:
    total: int = 0
    passed: int = 0
    failed: int = 0
    skipped: int = 0
    results: list[TResult] = field(default_factory=list)


class Engine:
    """api-flow 执行引擎。Agent 通过 `engine.run()` 调用，不暴露 CLI。"""

    def __init__(
        self,
        flow_path: Path | str,
        output_root: str = "api-flow-output",
        fail_fast: bool = False,
        db_config: dict[str, Any] | None = None,
    ) -> None:
        self._flow: FlowConfig = load_flow(flow_path)
        self._output_root = output_root
        self._fail_fast = fail_fast
        self._db_config = db_config or {}

    def run(self) -> RunResult:
        """执行全部测试用例。Agent 调用后拿到 RunResult 用于展示。"""
        result = RunResult()
        result.total = len(self._flow.test_cases)

        auth_injector = AuthInjector(auth_config_from_yaml({
            "type": self._flow.auth.type,
            "token_url": self._flow.auth.token_url,
            **self._flow.auth.extra,
        }))
        db = Database(
            host=self._db_config.get("host", "localhost"),
            user=self._db_config.get("user", "root"),
            password=self._db_config.get("password", ""),
            database=self._db_config.get("database", "test"),
            port=self._db_config.get("port", 3306),
        )
        http_client = HttpClient(base_url=self._flow.base_url)
        recorder = Recorder(self._flow.name, self._output_root)

        consecutive_failures = 0

        try:
            for tc in self._flow.test_cases:
                tc_result = self._run_one_with_retry(
                    tc, auth_injector, db, http_client, recorder, MAX_CASE_RETRIES
                )
                result.results.append(tc_result)

                if tc_result.status == "pass":
                    result.passed += 1
                    consecutive_failures = 0
                elif tc_result.status == "fail":
                    result.failed += 1
                    consecutive_failures += 1
                elif tc_result.status == "error":
                    result.failed += 1
                    consecutive_failures += 1
                else:  # skipped
                    result.skipped += 1

                # DB 致命错误 → 全局中断
                if db.consecutive_errors >= MAX_CONSECUTIVE_FAILURES:
                    break

                # 连续业务失败过多 → 全局中断
                if consecutive_failures >= MAX_CONSECUTIVE_FAILURES:
                    break

                # 遇错即停模式（仅针对业务失败）
                if self._fail_fast and tc_result.status in ("fail", "error"):
                    break
        finally:
            recorder.save_summary(result.results)
            http_client.close()
            db.close()

        return result

    def _run_one_with_retry(
        self,
        tc: TestCase,
        auth_injector: AuthInjector,
        db: Database,
        http_client: HttpClient,
        recorder: Recorder,
        max_retries: int,
    ) -> TResult:
        """执行单个用例，失败后 cleanup → 全新 prepare → 重试。"""
        last_result: TResult | None = None

        for attempt in range(max_retries):
            if attempt > 0:
                # 先执行 cleanup 回滚上次的脏数据
                if last_result and last_result.cleanup_sql:
                    try:
                        db.execute_cleanup(last_result.cleanup_sql, RuntimeVariables())
                    except Exception:
                        pass  # cleanup 失败忽略，下一轮用全新 UUID

            tr = self._run_one(tc, auth_injector, db, http_client, recorder)

            if tr.status == "pass":
                return tr

            # expect_error 用例：不重试
            if tc.expect_error:
                return tr

            last_result = tr

            # 不可重试的 DB 错误：不做用例级重试
            if tr.error and "不可重试" in str(tr.error):
                return tr

        # 重试耗尽
        if last_result:
            return last_result

        return TResult(tc_id=tc.id, scenario=tc.scenario, status="error", error="重试耗尽")

    def _run_one(
        self,
        tc: TestCase,
        auth_injector: AuthInjector,
        db: Database,
        http_client: HttpClient,
        recorder: Recorder,
    ) -> TResult:
        """执行单个测试用例（单次，不含 retry）。与 v0.1 核心逻辑相同。"""
        tr = TResult(tc_id=tc.id, scenario=tc.scenario, status="skipped")
        runtime = RuntimeVariables()

        token = auth_injector.get_token()
        runtime.set_auth_token(token)

        start = time.time()

        # Phase 1: Prepare
        try:
            db.execute_prepare(tc.prepare, runtime)
            tr.prepare_sql = tc.prepare
        except Exception as e:
            tr.status = "error"
            tr.error = f"prepare 失败: {e}"
            tr.duration_ms = (time.time() - start) * 1000
            recorder.record(tr)
            return tr

        # Phase 2: Run
        try:
            response = http_client.request(
                method=tc.request.method,
                url=tc.request.url,
                headers=tc.request.headers,
                body=tc.request.body,
                runtime=runtime,
            )
            tr.status_code = response.status_code
            tr.request_body = tc.request.body
            tr.response_body = response.body
            tr.duration_ms = (time.time() - start) * 1000

            if tc.expect_error:
                tr.status = "pass" if response.status_code >= 400 else "fail"
            else:
                tr.status = "pass" if 200 <= response.status_code < 300 else "fail"

            http_client.extract_to_runtime(response.body, tc.extract, runtime)

        except Exception as e:
            tr.status = "error"
            tr.error = f"请求失败: {e}"
            tr.duration_ms = (time.time() - start) * 1000
            recorder.record(tr)
            return tr

        # Phase 3: Cleanup
        try:
            substituted_cleanup = [substitute(stmt, runtime) for stmt in tc.cleanup]
            tr.cleanup_sql = substituted_cleanup
            db.execute_cleanup(substituted_cleanup)
        except Exception as e:
            pass  # cleanup 失败不改变 test case 结果

        recorder.record(tr)
        return tr
```

"""Agent 调用入口：Agent import 并调用 execute_flow()，不暴露 CLI。"""

from __future__ import annotations

from pathlib import Path
from typing import Any

from api_flow.engine import Engine, RunResult
from api_flow.templates import render_case_result, render_summary


def execute_flow(
    flow_dir: str,
    db_config: dict[str, Any] | None = None,
    fail_fast: bool = False,
) -> dict[str, Any]:
    """Agent 调用此函数执行整个 flow。

    Args:
        flow_dir: flow 目录路径（包含 flow.yaml）
        db_config: 数据库配置 dict，包含 host/user/password/database/port
        fail_fast: 遇错即停模式

    Returns:
        {
            "flow_name": str,
            "run_result": RunResult,
            "case_reports": list[str],     # 每条用例的 Markdown 报告
            "summary_report": str,         # 汇总 Markdown 报告
            "output_dir": str,             # 输出目录路径
        }
    """
    engine = Engine(
        flow_path=Path(flow_dir),
        output_root=Path(flow_dir) / "results",
        fail_fast=fail_fast,
        db_config=db_config or {},
    )
    run_result = engine.run()

    # 生成每条用例的模板化报告
    case_reports: list[str] = []
    for tr in run_result.results:
        report = render_case_result(
            tc_id=tr.tc_id,
            scenario=tr.scenario,
            branch=getattr(tr, "branch", ""),
            priority=getattr(tr, "priority", "medium"),
            status=tr.status,
            status_code=tr.status_code,
            duration_ms=tr.duration_ms,
            method=getattr(tr, "method", ""),
            url=getattr(tr, "url", ""),
            request_body=tr.request_body if hasattr(tr, "request_body") else None,
            response_body=tr.response_body,
            prepare_sql=tr.prepare_sql if hasattr(tr, "prepare_sql") else None,
            cleanup_sql=tr.cleanup_sql if hasattr(tr, "cleanup_sql") else None,
            error=tr.error,
        )
        case_reports.append(report)

    # 生成汇总报告
    result_dicts = [
        {
            "tc_id": tr.tc_id,
            "scenario": tr.scenario,
            "status": tr.status,
            "status_code": tr.status_code,
            "duration_ms": tr.duration_ms,
            "error": tr.error,
        }
        for tr in run_result.results
    ]
    summary_report = render_summary(engine._flow.name, result_dicts)

    return {
        "flow_name": engine._flow.name,
        "run_result": run_result,
        "case_reports": case_reports,
        "summary_report": summary_report,
        "output_dir": str(engine._output_root),
    }
```

name: api-flow
description: API 流程测试工具。自动分析 Java/Spring Boot 代码，生成多分支测试用例和输入数据，逐条确认后批量触发 API 流程，人工验证响应结果，自动清理数据。当用户需要测试 API 接口流程、生成多分支测试用例、批量执行接口调用、或需要自动准备测试数据并清理时使用此 skill。

# api-flow

自动化 API 流程测试工具：分析代码 → 生成用例 → 逐条确认执行 → 数据清理。

## 核心概念

- **analyze**（Phase 1，AI 负责）：gitnexus 分析代码 → 识别分支 → 生成 flow.yaml + tc-*.json + tc-*-{prepare,cleanup}.sql
- **execute**（Phase 2，Agent 调用 Python engine）：逐用例 checkpoint 确认 → prepare → request → cleanup → 展示结果
- 每条用例独立执行，不共享数据，不互相污染
- **纯对话式**——用户不需要记忆任何命令

## 前置要求

1. 项目已安装 gitnexus
2. Python engine 已安装：`cd skill/api-flow/cli && pip install -e ".[dev]"`
3. MySQL 数据库可访问（用于 prepare/cleanup）

---

## 工作流

### Phase 1: 分析代码，生成用例文件

用户只需提供 API 入口：
```
"帮我生成订单创建流程的测试用例"  →  Controller: com.example.controller.OrderController.createOrder()
"分析 POST /api/orders 的所有分支"  →  同上
```

**AI 执行 analyze 流程：**

1. **发现入口** — gitnexus 搜索 Controller 类和方法
2. **追踪调用链** — Controller → Service → Repository，最多 3 层
3. **识别分支** — 扫描 @Valid、if/else/switch、throw、自定义校验
4. **生成用例** — 每个分支一个用例，含中文场景描述
5. **生成 prepare SQL** — 保存为 `tc-*-prepare.sql`，带注释说明依赖表和 FK 顺序
6. **生成 cleanup SQL** — 保存为 `tc-*-cleanup.sql`，FK 逆序
7. **生成 request body** — 保存为 `tc-*.json`，变量引用用 `{{@var}}`
8. **生成 flow.yaml** — 编排引用，不含用例内容

**输出目录**: `docs/api-flow/<flow-name>/`

生成完成后展示摘要表格（设计规格书 §3.4），用户审阅修改后确认。

### Phase 2: 逐用例执行

用户确认后，Agent 开始逐条执行：

每条用例执行前展示 checkpoint（设计规格书 §4.2 格式）：
- 用例 ID + 场景 + 分支
- prepare SQL 影响哪些表
- request method + URL + body
- cleanup 概况

用户确认 → Agent 调用 `execute_flow()` → 展示结果 → 人工验证 → 下一条。

### 失败处理

- DB 可重试错误（死锁/断连）：自动退避重试 3 次 → 仍失败则全新 prepare 重试该用例（最多 3 次）
- DB 不可重试错误（语法/权限）：标记 FAILED，继续下一条
- HTTP 业务错误：标记 FAILED，继续下一条
- 连续 3 条用例失败：全局中断（基本确定 DB 挂了）

### Phase 3: 汇总报告

全部用例执行完毕后，输出 summary.md。
